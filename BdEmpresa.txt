CREATE DATABASE EMPRESA;

USE EMPRESA;

CREATE TABLE USUARIOS (
    CPF VARCHAR(11) NOT NULL,
    NOME VARCHAR(50) NOT NULL,
    NASCIMENTO DATE NOT NULL,
    ENDERECO VARCHAR(100),
    BAIRRO VARCHAR(30),
    CIDADE VARCHAR(30),
    UF VARCHAR(2),
    CELULAR VARCHAR(15),
    EMAIL VARCHAR(100),
    USUARIO VARCHAR(20) NOT NULL,
    SENHA VARCHAR(10) NOT NULL,

    PRIMARY KEY (CPF)
);
